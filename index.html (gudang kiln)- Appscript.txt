/* ════════════════════════════════════════════════════════════════
MONITORING STOK
════════════════════════════════════════════════════════════════ */
function renderTable() {
const q = (document.getElementById('search')?.value || '').toLowerCase().trim();
const g = document.getElementById('f-gudang')?.value || '';
const k = document.getElementById('f-kategori')?.value || '';
const s = document.getElementById('f-status')?.value || '';
const tbody = document.getElementById('stok-body');
if (!tbody) return;

const filtered = state.stock.filter(r => {
const low = +r.stok_awal <= +r.min_stok;
const matchQ = !q || r.nama_barang.toLowerCase().includes(q) || r.code_barang.toLowerCase().includes(q);
return matchQ && (!g || r.gudang === g) && (!k || r.kategori_barang === k)
&& (!s || (s === 'LOW' && low) || (s === 'OK' && !low));
});

const kritis = filtered.filter(r => +r.stok_awal <= +r.min_stok).length;
setT('stat-total', filtered.length);
setT('stat-low', kritis);
setT('stat-ok', filtered.length - kritis);

if (!filtered.length) {
tbody.innerHTML = `<tr><td colspan="7"><div class="empty-state"><div class="empty-state-icon">📦</div><h3>Tidak ada data ditemukan</h3><p>Coba ubah filter pencarian Anda</p></div></td></tr>`;
return;
}

tbody.innerHTML = filtered.map(r => {
const low = +r.stok_awal <= +r.min_stok;
const pct = r.min_stok > 0 ? Math.min(100, Math.round(+r.stok_awal / +r.min_stok * 100)) : 100;
return `<tr>
<td><span class="code-text">${escHtml(r.code_barang)}</span></td>
<td>
<div style="font-weight:600;font-size:.875rem;">${escHtml(r.nama_barang)}</div>
<div style="font-size:.7rem;color:var(--ink-400);margin-top:1px;">${escHtml(r.satuan_barang)}</div>
</td>
<td style="font-size:.82rem;">${escHtml(r.gudang)}</td>
<td><span class="badge badge-cat">${escHtml(r.kategori_barang)}</span></td>
<td style="text-align:right;">
<span style="font-family:var(--font-display);font-size:1.05rem;font-weight:700;color:${low?'var(--rose)':'var(--ink-900)'};">${r.stok_awal}</span>
<div style="width:60px;margin-left:auto;margin-top:3px;">
<div class="progress-bar"><div class="progress-fill" style="width:${pct}%;background:${low?'var(--rose)':'var(--green)'}"></div></div>
</div>
</td>
<td style="text-align:right;font-size:.82rem;color:var(--ink-400);">${r.min_stok}</td>
<td style="text-align:center;"><span class="badge ${low?'badge-low':'badge-ok'}">${low?'KRITIS':'AMAN'}</span></td>
</tr>`;
}).join('');
}

function auRow(inp) {
const tr = inp.closest('tr');
const sistem = +tr.dataset.sistem;
const aktual = +inp.value;
const sel = aktual - sistem;
const cell = tr.querySelector('.au-sel');
if (cell) {
cell.innerHTML = sel === 0 ? '—'
: sel > 0 ? `<span style="color:var(--green);font-weight:700;">+${sel}</span>`
: `<span style="color:var(--rose);font-weight:700;">${sel}</span>`;
}
inp.style.borderColor = sel === 0 ? 'var(--ink-200)' : sel > 0 ? 'var(--green)' : 'var(--rose)';
updateAuditSummary();
}

function updateAuditSummary() {

setT('au-sum-total', rows.length);

setT('au-sum-turun', turun);
setT('au-sum-sama', sama);
}

function makeTable(heads, rows, emptyMsg) {
if (!rows.length) return `<div class="empty-state" style="padding:3rem;"><div class="empty-state-icon">📋</div><h3>Tidak ada data</h3><p>${emptyMsg||''}</p></div>`;
const ths = heads.map(h => `<th style="text-align:left;padding:.75rem 1rem;font-size:.67rem;font-weight:700;letter-spacing:.07em;text-transform:uppercase;color:var(--ink-400);background:var(--ink-50);border-bottom:1px solid var(--ink-200);${h.th||''}">${h.label}</th>`).join('');
const trs = rows.map((r,i) => {
const bg = i%2===0 ? 'white' : 'var(--ink-50)';
const tds = heads.map(h => `<td style="padding:.75rem 1rem;border-bottom:1px solid var(--ink-100);font-size:.875rem;${h.td||''}">${r[h.key]??''}</td>`).join('');
return `<tr style="background:${bg};">${tds}</tr>`;
}).join('');
return `<div style="overflow-x:auto;border:1px solid var(--ink-200);border-radius:var(--r);"><table style="width:100%;border-collapse:collapse;min-width:500px;"><thead><tr>${ths}</tr></thead><tbody>${trs}</tbody></table></div>`;
}

function promptAdmin() {
pinBuffer = ''; renderDots();
document.getElementById('pin-error').innerHTML = '&nbsp;';
document.getElementById('pin-attempts').innerHTML = '';
document.getElementById('pin-overlay').classList.add('active');
document.body.style.overflow = 'hidden';
}
function closePinModal() {
document.getElementById('pin-overlay').classList.remove('active');
document.body.style.overflow = '';
pinBuffer = ''; renderDots();
}
document.addEventListener('click', e => { if (e.target === document.getElementById('pin-overlay')) closePinModal(); });
document.addEventListener('keydown', e => {
if (!document.getElementById('pin-overlay').classList.contains('active')) return;
if (e.key >= '0' && e.key <= '9') pinKey(+e.key);
else if (e.key === 'Backspace') pinDel();
else if (e.key === 'Escape') closePinModal();
});
function pinKey(n) {
if (pinLocked || pinBuffer.length >= 6) return;
pinBuffer += n; renderDots();
if (pinBuffer.length === 6) setTimeout(checkPin, 200);
}
function pinDel() {
if (pinLocked) return;
pinBuffer = pinBuffer.slice(0,-1); renderDots();
document.getElementById('pin-error').innerHTML = '&nbsp;';
}
function renderDots() {
for (let i=0; i<6; i++) {
const d = document.getElementById('pd'+i);
if (!d) continue;
d.classList.remove('filled','error');
if (i < pinBuffer.length) d.classList.add('filled');
}
}
function checkPin() {
if (pinBuffer === PIN_CORRECT) {
pinAttempts = 0; pinBuffer = '';
setTimeout(() => { closePinModal(); activateAdmin(); }, 250);
} else {
pinAttempts++;
for (let i=0; i<6; i++) { const d=document.getElementById('pd'+i); if(d){d.classList.remove('filled');d.classList.add('error');} }
document.getElementById('pin-error').textContent = 'PIN salah. Periksa kembali.';
if (pinAttempts >= 5) {
pinLocked = true; let cd = 30;
const at = document.getElementById('pin-attempts');
const tick = () => { if(at) at.innerHTML = `<span class="pin-badge">🔒 Terkunci ${cd} detik</span>`; };
tick();
pinTimer = setInterval(() => {
cd--; tick();
if (cd <= 0) { clearInterval(pinTimer); pinLocked=false; pinAttempts=0; pinBuffer=''; renderDots(); document.getElementById('pin-error').innerHTML='&nbsp;'; document.getElementById('pin-attempts').innerHTML=''; }
}, 1000);
} else {
const rem = 5 - pinAttempts;
const at = document.getElementById('pin-attempts');
if (at) at.innerHTML = `<span class="pin-badge">⚠ ${pinAttempts} salah · ${rem} sisa</span>`;
}
setTimeout(() => {
pinBuffer = ''; renderDots();
for (let i=0;i<6;i++){const d=document.getElementById('pd'+i);if(d)d.classList.remove('error');}
}, 650);
}
}

function showToast(msg, success=true) {
const t=document.getElementById('toast');
if(!t) return;
t.textContent = msg;
t.style.background = success
? 'linear-gradient(135deg,var(--green) 0%,#059669 100%)'
: 'linear-gradient(135deg,var(--rose) 0%,#C0142A 100%)';
t.classList.add('show');
clearTimeout(t._timer);
t._timer = setTimeout(() => t.classList.remove('show'), 4500);
}
function escHtml(str) {
if (str===null||str===undefined) return '';
return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
function escAttr(str) {
if (str===null||str===undefined) return '';
return String(str).replace(/"/g,'&quot;').replace(/'/g,'&#39;');
}
